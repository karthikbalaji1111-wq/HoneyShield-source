     kubectl -n {{ .Release.Namespace }} get svc {{ include "hs.name" . }}-edge

     kubectl -n {{ .Release.Namespace }} exec -it deploy/{{ include "hs.name" . }}-backend -- python -m app.bootstrap --email you@example.com

     HS_VERIFY_ADMIN_PASSWORD=... python3 scripts/verify_deployment.py --control-host {{ .Values.hosts.control }} \
       --ingest-host {{ .Values.hosts.ingest }} --admin-email you@example.com --totp-secret-file admin.totp
